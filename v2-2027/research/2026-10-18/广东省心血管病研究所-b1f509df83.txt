function handleLinkClick(event, type) {
    event = event || window.event;
    var target = event.target || event.srcElement;

    var link = findClosestLink(target);
    if (link) {
        var href = link.getAttribute('href');
        var label = link.textContent.trim();

        gaTrackerOutboundLink({ eventCategory: type, eventAction: 'click', eventLabel: label });
    }
}

function findClosestLink(element) {
    while (element && element.nodeName !== 'A') {
        element = element.parentNode;
    }
    if (element) {
        return element;
    }
    return null;
}
document.addEventListener('DOMContentLoaded', function () {
    const topNavSearch = document.getElementById('top-nav-search');
    const searchCancel = document.querySelector('.search-cancel');
    const searchInput = document.querySelector('.search-input');
    const realQuery = document.querySelector('.realQuery');
    const searchForm = document.querySelector('.site-search');

    // 点击搜索图标：显示输入框（添加 active 类）
    if (topNavSearch) {
        topNavSearch.addEventListener('click', function (event) {
            event.stopPropagation();
            document.querySelector('.top-nav-wrapper').classList.add('active');
            searchInput.focus();
        });
    }

    // 点击取消按钮：移除 active 类
    if (searchCancel) {
        searchCancel.addEventListener('click', function (event) {
            document.querySelector('.top-nav-wrapper').classList.remove('active');
        });
    }

    // 点击文档其他地方：移除 active 类
    document.addEventListener('click', function (event) {
        const wrapper = document.querySelector('.top-nav-wrapper');
        if (wrapper && !wrapper.contains(event.target)) {
            wrapper.classList.remove('active');
        }
    });

    // 提交表单前处理搜索词
    if (searchForm) {
        searchForm.addEventListener('submit', function (e) {
            e.preventDefault(); // 阻止默认提交

            const val = searchInput.value.trim();

            if (!val) {
                alert("请输入关键词");
                return;
            }

            // 设置 realQuery 的值为关键词 + site:xxx
            realQuery.value = val + ' site:yz.chsi.com.cn';

            // 手动提交表单
            searchForm.submit();
        });
    }
});

 广东省心血管病研究所2027年硕士研究生招生简章 

 
     广东省心血管病研究所创办于 1958年，为广东省人民医院（广东省医学科学院）下设直属机构，是1981年国务院批准的首批博士和硕士学位授予单位之一，2018年获批临床医学硕士专业学位授权点，2021年获批临床医学一级学科博士学位授权点，依托广东省人民医院（广东省医学科学院）医教研平台开展研究生培养工作，具备临床医学全部二级学科的研究生招生资格。经过几代人的共同努力，医院现已成为集医疗、教学、科研、保健于一体的大型现代化三级甲等综合型医院，在国内外享有盛誉。医院综合实力位居全国前列，国考绩效3次获评A++、稳居全国前1%。荣获2022-2025年度全国医疗机构“最佳雇主公立医院10强”和“最受大学生欢迎10强”。 一、招生计划 广东省心血管病研究所（以下简称 “研究所”）研究生招生代码为88911。 2027 年面向全国拟招收 硕士研究生 120 人 （ 含推荐免试生 3人 ） ，其中学术学位 45 人，专业学位 75 人 。 实际招生数将 以 国家正式下达的招生计划 为准 ，录取时将根据招生计 划和生源情况，对专业招生计划数进行适当调整 （后续会更新各专业招收统考计划，请考生关注我 院官网医教研研究生管理模块 ） 。 二、学制及培养 （一）学制和学习年限 硕士研究生基本学制为 3年，最长学习年限不超过5年。 （二）培养方式 全日制 培养模式。 全日制研究生 需符合 国家研究生招生规定，通过研究生入学考试录取 ， 并 在基本修业年限或者 心研所 规定 的学习 年限内，全脱产在 我院完成学业 。 （三）学位类型 1.学术学位：以培养从事基础研究或应用基础研究人员为目标，侧重于学术理论水平和实验研究能力的培养，学生毕业时授予 临床医学 学术型学位 。 2.专业学位：以培养高级应用型 临床 人才为目标，侧重于从事 临床 实践工作 实践 能力的培养 。 临床医学专业学位 研究生教育全部实行与住院医师规范化培训相结合的培养模式 。 3. 论文实验、实践环节：由导师根据培养计划安排。   三 、 报考条件 （一）基本条件 1.拥护中国共产党的领导，遵守中华人民共和国宪法和法律，诚实守信。 2.身体健康状况符合国家和我院规定的体检要求。 （二）学历条件 国家承认学历的普通全日制本科五年制及以上学制毕业生。考生 须在 录取当年 9月1日 前 ， 取得国家承认的本科毕业证书 、 学位证书 ， 或教育部留学服务中心出具的《国（境）外学历学位认证书》，否则录取资格无效。 （ 三） 专业限制条件 在满足上述条件的同时，还须同时满足本科毕业专业和学位要求。 1. 临床医学学术学位（专业代码 1002开头） ： 各专业 仅 接受本科专业为医学门类考生报考。 2. 临床医学专业学位 （ 专业 代码 1051 开头 ） ： ①除精神病与精神卫生学、放射影像学、超声医学、核医学、临床检验诊断学、眼科学、麻醉学、放射肿瘤学、儿科学以外 的其他 专业 ： 只 接收全日制本科为五年及以上学制西医临床医学专业（医学学位）的考生报考。   ②精神病与精神卫生学： 只 接收全日制本科为五年及以上学制西医临床医学 和 精神医学（医学学位）专业的考生报考； ③放射影像学、超声医学、核医学： 只 接收全日制本科为五年及以上学制西医临床医学 和 医学影像学（医学学位）专业的考生报考； ④临床检验诊断学： 只 接收全日制本科西医临床医学 和 医学检验（医学学位）专业的考生报考； ⑤眼科学： 只 接收全日制本科为五年及以上学制西医临床医学 和 眼视光医学（医学学位）专业的考生报考； ⑥麻醉学： 只 接收全日制本科为五年及以上学制西医临床医学 和 麻醉学（医学学位）专业的考生报考。 ⑦放射肿瘤学： 只 接收全日制本科为五年及以上学制西医临床医学 和 放射医学（医学学位）专业的考生报考。 ⑧儿科学： 只 接收全日制本科为五年及以上学制西医临床医学 和 儿科学（医学学位）专业的考生报考。 3. 临床医学专业学位研究生招生对象为符合医师资格考试报考条件规定专业的应届或往届本科毕业生 ； 入学前已获得临床住院医师规范化培训合格证书人员 ， 不得报考临床医学硕士专业学位研究生。 四、报名程序 （一）网上报名 考生应在规定时间登录 “ 中国研究生招生信息网 ”（网址：https://yz.chsi.com.cn）参加报名， 网上报名时间为 2026年10月15日至10月24日，每天9:00—22:00。网上报名要求以教育部的规定为准。 （二）网上确认 时间和具体要求以考生报名所选择的报考点的规定为准。 （三）资格审查 我院心研所将对考生的报考信息进行全面审查，因不符合报考条件及相关政策要求的，后果由考生本人承担。 （四）注意事项 1.考生要完整准确填写个人信息，应认真了解并严格按照报考条件及相关政策要求选择填报志愿。因不符合报考条件及相关政策要求，造成后续不能网上确认、考试、复试或录取的，后果由考生本人承担。 2.考生报名时只能填报一个专业。 3.未通过学历（学籍）校验的考生应及时到学籍学历权威认证机构进行认证，并将认证报告交报考点和我院心研所核验。 4.报考点选择：报考点即考试地点，以就近方便为原则，报考我院心研所的考生，只要满足报考点要求均可选择，并在该报考点参加考试，网报选择报考点时请仔细查看其要求或咨询当地报考点。 （五）网上确认 1.网上确认时间：按照各报考点通知。 2.网上确认要求： （ 1）所有考生均应在规定时间内按报考点要求核对并确认其网上报名信息，逾期不再补办。 （ 2）考生网上确认应提交本人居民身份证、毕业证书(应届生持学生证)及考点规定其他材料，由报考点工作人员进行核对。 （ 3）未通过网上学历（学籍）校验的考生，在确认时应提供学历（学籍）认证报告。 （ 4）所有考生均应对本人网上报名信息进行认真核对并确认。报名信息经考生确认后一律不作修改，因填写错误引起的一切后果自行承担。 （ 5）考生应按规定缴纳报考费。 （ 6）考生应按报考点规定配合采集本人图像等相关信息。 （ 7）考生网上报名成功后，应通过定期查阅所在省教育招生考试机构、报考点、主动了解考试安排、相关要求等事项，积极配合完成相关工作。 五、考试 （ 一 ） 初试时间： 2026年12月19-20日（具体时间以中国研究生招生信息网公布内容为准）。 （ 二 ）入学考试分初试和复试    初试由教育 部统一组织，实行全国统一考试。考生应当在考前十天左右，凭网报用户名和密码登录 “研招网”自行下载打印的《准考证》。考生凭有效居民身份证参加初试和复试。 （ 三） 初试地点 ： 由考生 自行 选定的报考点负责安排，详见《准考证》或咨询所选报考点。 （四） 初试方式 ： 均为笔试。 （五） 初试考试科目详见《 广东省心血管病研究所 2027 年招收硕士研究生专业目录》，全国统考各科考试时间均为 3小时。 （ 六 ）初试成 绩查询：教育部 将 在规定时间内开放初试成绩查询通道，考生需登录中国研究生招生信息网，使用学信网账号查询成绩。若对评卷结果有异议，可按教育部相关考务文件规定的程序申请成绩复查。 （ 七 ） 复试及调剂等工作安排将严格遵照国家相关要求执行，以届时的通知为准。   六、录取 （ 一）心研所 根据国家下达的招生计划， 结合 考生的综合成绩（含初试和复试成绩） 及 思想政治表现 等 ， 确定拟录取名单。复试不合格者，不予录取。 （ 二） 硕士研究生按就业方式分为定向就业 与 非定向就业 两类。 报考定向就业 的 考生 ， 须与定向就业单位签订定向培养合同，并 按规定时间 将合同 提交至心研所。 请报考定向就业 的 考生 务必充分 了解相关政策，因 个人 填报失误 导致无法 入学等后果，由考生 自行 承担 责任 。 （三） 拟录取考生须参加 我院心研 所组织的招生体检，身体健康状况 须 符合国家及 心研 所规定的体检要求，考生不得隐瞒个人病情；若 身体状况 不符合要求，将取消其录取资格。 （四） 心研所计划 于 2027 年 7月发放录取通知书，9月初组织新生入学资格复审。若 新生 不符合相关要求，将取消入学资格。   七、费用 （ 一）学费 1. 全日制非定向临床医学学术学位硕士研究生： 8000元 / 年。 2. 全日制非定向临床医学专业学位硕士研究生 ： 12000元/年 。 3. 全日制定向培养硕士研究生 ： 20000元/年。 （ 二）住宿费 按有关规定的标准执行。 八、 奖助学金政策 具体奖助学金政策按照《广东省人民医院研究生奖助学金管理规定》（粤医〔 2023〕47号）发放。 九、 其他说明 1.考生本人对网上报名信息需要进行认真核对并确认，经考生确认的报名信息在考试、复试及录取阶段一律不作修改，因考生填写错误引起的一切后果由其自行承担。 2.考生初试成绩、复试安排 、拟录取名单等重要信息，均将在   “广东省人民医院官网 - 医教研动态 - 研究生管理 - 研究生招生” 栏目发布，供考生查询，不另行下发书面通知。 3 .考生报名时 ， 只能填报一个专业 ， 复试时经师生互选确定导师。 4.有关我院心研所硕士研究生报名、考试、复试、录取等信息，均在我院官网医教研模块发布，其他渠道信息非我院发布的不予解释，请谨防上当。以上内容如遇国家政策调整或我院相关安排变化将另文说明，解释权归属广东省人民医院广东省心血管病研究所。未尽事宜以教育部《2027年全国硕士研究生招生工作管理规定》等相关文件为准。 十、 联系方式    1.  广东省心血管病研究所办公室 ：
