    function getCategoryList(pid) {
        // 读取时检查过期
        const tempData = JSON.parse(localStorage.getItem('tempData'+pid));
        if (tempData && Date.now() - tempData.timestamp < 60 * 1000) {
            console.log('缓存有效');
            console.log(tempData.data);
            let data = tempData.data;
            let html = "";
            let wap = "";
            data.forEach(item => {
                // 假设接口返回字段：title（标题）、date（日期字符串如 "09-25"）、url（可选）
                const title = item.c_name;
                const url = item.url;
                const children = item.children;
                console.log(children);
                let id = item.id;
                let cid = "91";
                if (item.children.length>0){
                    html += '<div class="nav_er_li on san">';
                    html += '<a href="javascript:;">'+title+'</a>'
                    html += '<div class="nav_er_ul1">';
                    for (let j = 0;j<item.children.length;j++){
                        let title2 = item.children[j].c_name;
                        let url2 = item.children[j].url;
                        let childId = item.children[j].id;
                        console.log(childId);
                        if (cid==childId){
                            html +='<div class="nav_er_li1 on">';
                        } else {
                            html +='<div class="nav_er_li1">';
                        }
                        html += '<a href="'+url2+'">'+title2+'</a>' +
                            '</div>';
                    }
                    html += '</div>';
                } else {
                    if (cid==id){
                        html += '<div class="nav_er_li on">';
                    } else {
                        html += '<div class="nav_er_li">';
                    }

                    html += '<a href="'+url+'">'+title+'</a>';

    function getLinkList(type_id, sort) {
        $.ajax({
            url: "/index/Api/getLinkList",
            data: { type_id: type_id, show_style: sort },
            dataType: "json",
            type: "GET",
            success: function (res) {
                const list = res.list || [];
                let html = '';

                if (list.length === 0) {
                    html = '<div class="main3_1_img_li"><a href="#">暂无数据</a></div>';
                } else {
                    list.forEach(item => {
                        // 假设字段：title（标题）、create_time 或 publish_time（时间戳）
                        const title = item.name || '无标题';
                        const url = item.url || '#';
                        html += '<li>' +
                            '<a href="'+url+'" target="_blank">' + title+'</a>' +
                            '</li>'
                    });
                }
                $('#link').html(html);
            },
            error: function (xhr) {
                console.error('加载失败:', xhr.statusText);
                $('#link').html('<li><a href="#">加载失败</a></li>');
            }
        });
    }
