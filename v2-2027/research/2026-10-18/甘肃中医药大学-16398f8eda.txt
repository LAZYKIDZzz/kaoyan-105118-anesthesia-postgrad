    function getArticleList(cid, num) {
        $.ajax({
            url: "/index/Api/getArticleList",
            data: {
                cid: cid,
                num: num,
            },
            dataType: "json",
            type: "GET",
            success: function (res) {
                let data = res.list || [];
                let html = '';

                if (data.length === 0) {
                    html = '<div class="main_li"><a href="#">暂无数据</a></div>';
                } else {
                    data.forEach(item => {
                        // 假设接口返回字段：title（标题）、date（日期字符串如 "09-25"）、url（可选）
                        const title = item.title || '无标题';
                        const date = formatTimestamp(item.updatetime,'MM-DD') || '--';
                        const url = item.url || '#';
                        html += '<div class="main_li">'+
                            '<a href="'+url+'">' +
                            '<h3>'+title+'</h3>'+
                            '<span>'+date+'</span>'+
                            '</a>'+
                            '</div>';
                    });
                }
                $('#news').html(html);
            },
            error: function (xhr) {
                console.error('获取文章列表失败:', xhr.statusText);
                $('#news').html('<div class="main_li"><a href="#">加载失败，请稍后重试</a></div>');
            }
        });
    }

    /**
     * 将时间戳转换为指定格式的日期字符串
     * @param {number} timestamp - 时间戳（支持秒或毫秒）
     * @param {string} format - 日期格式模板，例如 'YYYY-MM-DD'、'MM/DD/YYYY HH:mm:ss'
     * @returns {string} 格式化后的日期字符串
     */
    function formatTimestamp(timestamp, format = 'YYYY-MM-DD') {
        // 如果是 10 位时间戳（秒），则转为毫秒
        if (String(timestamp).length === 10) {
            timestamp = timestamp * 1000;
        }

        const date = new Date(timestamp);
        if (isNaN(date.getTime())) {
            return 'Invalid Date';
        }

        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, '0'); // 月份从0开始
        const day = String(date.getDate()).padStart(2, '0');
        const hours = String(date.getHours()).padStart(2, '0');
        const minutes = String(date.getMinutes()).padStart(2, '0');
        const seconds = String(date.getSeconds()).padStart(2, '0');

        return format
            .replace(/YYYY/g, year)
            .replace(/YY/g, String(year).slice(-2))
            .replace(/MM/g, month)
            .replace(/DD/g, day)
            .replace(/HH/g, hours)
            .replace(/mm/g, minutes)
            .replace(/ss/g, seconds);
    }
