    function checkValues(){
		var url = $("#securl").val();
        var keyword = $("input[name='keyword']").val();
		if (keyword === '请输入关键字') {
			keyword = ''; 
		}
		if (keyword !== '') {
			keyword = encodeURI(encodeURI(keyword));
			url = url + "&keyword=" + keyword;
		}
        window.open(url);
    }

												 学校召开新学期研究生教学工作会议 
												 09-14 
											 
											 9月10日，学校召开2026-2027学年秋学期研究生教学工作会议，统筹部署新学期研究生教学各项重点工作。校党委常委、副校长邱志海主持会议并作总结讲话。邱志海肯定了学校研究生教育阶段性工作成效。自学校第九次党代会召开以来，学校始终将研究生教育作为高层次人才培养的核心战略抓手，各项教学建设与人才培养工作稳步推进、成效显著。围绕新学期研究生教学高质量发展，邱志海要求，全校上下要坚守立德树人根本任务，抓实课堂教学主阵地建设，筑牢研究生教育育人根基；聚焦专业学位研究生培养特色，深耕产教融合育人模式，全面提升应用型高层次医学人才培养水平；加速推进研究生教学数字化改革，迭代升级智慧教学模式与能力，赋能教学提质增效；从严抓实教学常态化管理，完善全流程质量监控闭环体系，夯实教育教学质量底线。会议明确，各教学单位及全体授课教师要牢固树立质量核心意识，主动扛起育人主体责任，把思想政治教育全方位融入研究生培养各环节。持续深化教风学风专项建设，扎实推进课程思政落地、案例教学革新、AI智慧课程打造及数字化教学平台提质升级，以高质量研究生教育发展成效，为学校“双一流”创建工作筑牢人才培养支撑。会上，研究生工作部通报并部署新学期研究生教学整体安排，教育质量评估中心督导专家复盘分析前期研究生教学督导工作情况、梳理现存问题与改进方向。部分教师代表结合自身教学实践，围绕课程体系建设、教学模式创新、育人质量提升等核心内容开展交流分享，探索教学改革新思路、新举措。学校相关职能部门、各教学单位负责人、相关工作人员及研究生教育督导专家参会。（研究生工作部） 
										 
									 
								 
								
								 
									 
										 
											 
										 
										 
											 
												 学校举行2026年博士研究生国家奖学金评审答辩会 
												 10-15 
											 
											 10月9日，学校举办2026年博士研究生国家奖学金评审答辩会。校党委常委、副校长邱志海出席会议并讲话。邱志海表示，研究生国家奖学金是国家面向优秀研究生设立的重要荣誉奖项，是落实立德树人根本任务、完善学生资助体系、提高研究生教育培养质量的有效手段。近年来，我校研究生国家奖学金评选成效显著，培育了一批优秀人才，为国家的创新发展提供了重要人才支撑。他对同学们提出三点希望：一要坚定理想信念、筑牢思想根基，以国家需求为导向，在实践中提高学术能力和专业素质；二要坚持学术创新、强化科研能力，扩展学术视野，培养创新思维，增强创新能力；三要坚守学术情操，对标榜样先锋，以学术诚信为治学之本，恪守学术规范，持续锤炼本领、突破自我。经个人申报、培养单位推荐、学校研究生工作部初审，共有54名优秀博士研究生参与本次答辩。答辩分临床学科组和基础学科组同步进行。学校将根据评审结果，从答辩候选人中最终评选出28名博士研究生获得国家奖学金。答辩过程中，同学们从研究背景、科技创新、课题研究、学术交流及代表性成果等方面进行汇报展示，评审过程严格遵循公开、公平、公正的原则。本次答辩会既集中检验了博士研究生培养质量和综合素质，也为博士研究生搭建了交流平台。学校将以此为契机，进一步发挥奖助体系育人功能，推动研究生教育高质量发展。（研究生工作部） 
										 
									 
								 
								
								 
									 
										 
											 
										 
										 
											 
												 2026级研究生新生迎来“研途第一课” 
												 09-14 
											 
											 9月7日，研究生工作部组织开展2026级研究生新生入学教育活动，围绕校史校情、心理健康、公派留学、研究生教育政策、科研创新、学术伦理、反诈安全等八个专题展开，为研究生新生量身打造“研途第一课”，迈好研究生阶段的第一步。徐叔云讲席教授陶芳标以《研究生科研选题专业取向与创新思路》为题，从师生共进、专业取向、创新路径三层面，鼓励新生主动融入导师课题，完成从知识接受者到创造者的转变。徐叔云讲席教授魏伟以《坚持四个面向，努力科技探索》为题，结合自身药理研究经历，勉励新生将学术追求与国家战略同频共振，把论文写在守护人民健康的实践中。科技产业部执行部长孟晓明聚焦科研诚信与伦理，通过界定学术不端、剖析典型违规案例，为新生划清科研底线。肥西县公安局反诈民警张晓婉结合真实案例，传授电信诈骗识别与防范方法，增强新生安全意识。校图书档案馆直属党支部书记张芳以百年校庆为坐标，讲述学校从黄浦江畔辗转至江淮之滨的办学历程，传递安医人薪火相传的精神力量。精神卫生与心理科学学院院长朱春燕结合临床数据与典型案例，系统讲授研究生心理调适策略，引导新生建立积极心理防御机制。国际学院党委副书记吴建章详解国家公派留学的政策背景、选派类别、申报要点及常见误区，提供务实操作指引。研究生工作部副部长温家根围绕日常管理、培养流程、学术规范、奖助体系及导学关系等，帮助新生构建清晰的学业架构认知。研究生新生入学教育既是研究生培养阶段的起点，也是强化思想引领、落实立德树人根本任务的关键抓手。下一步，研工部将围绕理想信念、专业素养等主题，持续深化入学教育内涵，指导各培养单位开展形式多样的入学教育活动，切实为2026级研究生顺利开启学业新征程奠定坚实基础。本次活动采用线下集中与线上学习相结合的方式开展，共有八个专题课程内容全部上线研究生网络教学平台，并将其纳入入学先导必修课程，供全体研究生新生学习。研究生工作部相关工作人员、部分2026级研究生新生代表参加本次活动。（研究生工作部 惠亚茹）

铭记伟大历史胜利，凝聚正义和平力量。9月3日上午，纪念中国人民抗日战争暨世界反法西斯战争胜利80周年大会在北京天安门广场隆重举行，中共中央总书记、国家主席、中央军委主席习近平发表重要讲话并检阅受阅部队。校领导班子以及全校师生医护员工，在各校区和各医院的报告厅、会议室、党员活动室、科室等场所集中收看大会直播盛况，共同见证这一庄严而伟大的时刻。习近平总书记在大会上发表的重要讲话振奋人心，让全体师生备受鼓舞；盛大的阅兵仪式气势恢宏，自豪之情油然而生。我校“中国人民抗日战争胜利80周年”纪念章获得者、离休老同志张秀亭，与第一附属医院北区的医护员工一起观看了大会直播，感受新时代中国军队的强大和国家的繁荣昌盛。“我是刘伯承部队的兵，刘元帅亲自拍着我说‘好样的’。那时候我才18岁。”观看过程中，张秀亭老人神情专注，不时提起对那段烽火岁月的深刻记忆，语气里满是自豪。校人力资源部（教师工作部）部长凌春笋谈道：“缅怀先烈当铭记历史，守卫和平当砥砺前行。新时代新征程，我们将坚定不移以习近平新时代中国特色社会主义思想为指导，传承和弘扬伟大抗战精神，牢固树立‘人才是第一资源’理念，坚持尊重劳动、尊重知识、尊重人才、尊重创造，全方位支持人才、帮助人才，千方百计造就人才、成就人才，不断激发广大人才干事创业积极性、主动性、创造性，持续为服务人才强国、教育强国建设贡献力量。”本科生工作部部长郭勇说：“伟大抗战精神所蕴含的爱国情怀、民族气节、英雄气概与必胜信念，也是蕴含在我们百年安医精神中最深刻的内容，是我们做好学生思政工作最磅礴的力量源泉。我们将把握教育契机，引导安医学子铭记历史，锤炼‘厚德’品格，厚植爱国情怀。同时，以穿透力强的思政品牌活动引导学生将个人理想熔铸于国家发展及学校‘双一流’建设目标中，在追求卓越中展现‘有为’担当，在探索未知中激扬‘创新’锐气，在兼容并蓄中彰显‘开放’格局；聚焦关键小事，践行‘响应式服务’工作理念，共同把学校建设成为大家的“幸福”家园。”研究生工作部部长潘莉莉感慨道：“这场阅兵是抗战胜利80周年的纪念，更是伟大抗战精神的传承。各个阅兵方队承载的烽火功勋，让我深切忆及抗战时期我校辗转多地，在艰难环境中坚持办学、救死扶伤，以医学为武器守护生命的那段厚重历史。未来工作中，我们将深挖百年校史中医学前辈在战火中挽救生命、传承医学火种的壮举，引导研究生铭记历史、赓续血脉，把伟大抗战精神与医学使命相融，以百年安医精神为范，努力成长为兼具家国情怀与国际视野的医学人才，为守护人民健康、助力国家发展贡献青春力量。”离退休教职工党委书记谢敏说：“作为一名老干部工作者，我们应始终铭记革命先辈的历史功绩，大力弘扬伟大的爱国主义精神和抗战精神。要以高度的政治自觉和真挚的情感，带着感情、带着责任，关心照顾好老战士、老同志，用心用情做好各项服务保障工作，切实帮助他们解决实际困难，让他们安享幸福晚年，共享发展成果。同时，我们要广泛宣传和弘扬老战士、老同志的英雄事迹和革命精神，让他们切身感受到党的温暖和社会的尊崇。”校团委书记贾伟华表示：“作为新时代的教育工作者和团学工作者，我们将牢记为党育人、为国育才的初心使命，坚持以团员青年为本，直接联系、服务、引导团员青年，注重发挥团员青年主体作用，带领团员青年深入学习抗战历史，传承和弘扬伟大抗战精神，进一步激发他们的爱国热情和奋斗精神，为实现中华民族伟大复兴贡献青年一代的力量。”保卫处副处长、武装部副部长薛政认为，从徒步方队的威武英姿到装备方队的科技硬核，盛大的阅兵仪式就是最生动的国防“教材”。“我们将以此为契机，持续加强军事理论课建设，积极开展伟大抗战精神宣传教育、国家安全形势讲座等活动，引导更多学子主动关注国防、参与国防建设。我们也将学习受阅部队展现出的良好作风并融入校园安全管理全过程，从日常巡逻的‘全覆盖’、隐患排查的‘无死角’，到应急处置的‘快响应’，用坚守与付出为师生营造安心的学习工作环境，为学校高质量发展和‘双一流’创建筑牢安全根基。”马克思主义学院院长柳泽民教授说：“习近平总书记在讲话中指出，‘中国人民坚定站在历史正确一边、站在人类文明进步一边，坚持走和平发展道路，与各国人民携手构建人类命运共同体’。这充分表明了中国共产党和中国人民以史为鉴、面向未来的历史自觉和历史担当。历史决不能忘记，更不能重演。我们铭记历史、缅怀先烈，是为了更好地珍爱和平、开创未来。作为一名思政课教师，我们将认真学习领会习近平总书记重要讲话精神，向广大青年学生讲好中国人民的抗战故事，教育引导其传承和弘扬伟大抗战精神，牢固树立正确的历史观，自觉积极投身实现中华民族伟大复兴的历史伟业。”校退休老同志、关工委委员陈海燕表示，作为一名新四军老战士的后代、一名退役军人，心情无比激动：“盛大阅兵让我深受教育、备受鼓舞、深感自豪。我要弘扬伟大抗战精神， 			function addEvent(obj, evtType, func, cap) {
				cap = cap || false;
				if (obj.addEventListener) {
					obj.addEventListener(evtType, func, cap);
					return true;
				} else if (obj.attachEvent) {
					if (cap) {
						obj.setCapture();
						return true;
					} else {
						return obj.attachEvent("on" + evtType, func);
					}
				} else {
					return false;
				}
			}

			function getPageScroll() {
				var xScroll, yScroll;
				if (self.pageXOffset) {
					xScroll = self.pageXOffset;
				} else if (document.documentElement && document.documentElement.scrollLeft) {
					xScroll = document.documentElement.scrollLeft;
				} else if (document.body) {
					xScroll = document.body.scrollLeft;
				}
				if (self.pageYOffset) {
					yScroll = self.pageYOffset;
				} else if (document.documentElement && document.documentElement.scrollTop) {
					yScroll = document.documentElement.scrollTop;
				} else if (document.body) {
					yScroll = document.body.scrollTop;
				}
				arrayPageScroll = new Array(xScroll, yScroll);
				return arrayPageScroll;
			}

			function GetPageSize() {
				var xScroll, yScroll;
				if (window.innerHeight && window.scrollMaxY) {
					xScroll = document.body.scrollWidth;
					yScroll = window.innerHeight + window.scrollMaxY;
				} else if (document.body.scrollHeight > document.body.offsetHeight) {
					xScroll = document.body.scrollWidth;
					yScroll = document.body.scrollHeight;
				} else {
					xScroll = document.body.offsetWidth;
					yScroll = document.body.offsetHeight;
				}
				var windowWidth, windowHeight;
				if (self.innerHeight) {
					windowWidth = self.innerWidth;
					windowHeight = self.innerHeight;
				} else if (document.documentElement && document.documentElement.clientHeight) {
					windowWidth = document.documentElement.clientWidth;
					windowHeight = document.documentElement.clientHeight;
				} else if (document.body) {
					windowWidth = document.body.clientWidth;
					windowHeight = document.body.clientHeight;
				}
				if (yScroll < windowHeight) {
					pageHeight = windowHeight;
				} else {
					pageHeight = yScroll;
				}
				if (xScroll < windowWidth) {
					pageWidth = windowWidth;
				} else {
					pageWidth = xScroll;
				}
				arrayPageSize = new Array(pageWidth, pageHeight, windowWidth, windowHeight)
				return arrayPageSize;
			}

				var winscroll = getPageScroll();
				AdMoveConfig.ScrollX = winscroll[0];
				AdMoveConfig.ScrollY = winscroll[1];
			}

			addEvent(window, "scroll", AdMoveConfig.Scroll);

			function AdMove(id) {
				if (!AdMoveConfig.IsInitialized) {
					AdMoveConfig.Resize();
					AdMoveConfig.IsInitialized = true;
				}
				var obj = document.getElementById(id);
				obj.style.position = "absolute";
				var W = AdMoveConfig.MoveWidth - obj.offsetWidth;
				var H = AdMoveConfig.MoveHeight - obj.offsetHeight;
				var x = W * Math.random(),
					y = H * Math.random();
				var rad = (Math.random() + 1) * Math.PI / 6;
				var kx = Math.sin(rad),
					ky = Math.cos(rad);
				var dirx = (Math.random() < 0.5 ? 1 : -1),
					diry = (Math.random() < 0.5 ? 1 : -1);
				var step = 1;
				var interval;
				this.SetLocation = function(vx, vy) {
					x = vx;
					y = vy;
				}
				this.SetDirection = function(vx, vy) {
					dirx = vx;
					diry = vy;
				}
				obj.CustomMethod = function() {
					obj.style.left = (x + AdMoveConfig.ScrollX) + "px";
					obj.style.top = (y + AdMoveConfig.ScrollY) + "px";
					rad = (Math.random() + 1) * Math.PI / 6;
					W = AdMoveConfig.MoveWidth - obj.offsetWidth;
					H = AdMoveConfig.MoveHeight - obj.offsetHeight;
					x = x + step * kx * dirx;
					if (x < 0) {
						dirx = 1;
						x = 0;
						kx = Math.sin(rad);
						ky = Math.cos(rad);
					}
					if (x > W) {
						dirx = -1;
						x = W;
						kx = Math.sin(rad);
						ky = Math.cos(rad);
					}
					y = y + step * ky * diry;
					if (y < 0) {
						diry = 1;
						y = 0;
						kx = Math.sin(rad);
						ky = Math.cos(rad);
					}
					if (y > H) {
						diry = -1;
						y = H;
						kx = Math.sin(rad);
						ky = Math.cos(rad);
					}
				}
				this.Run = function() {
					var delay = 10;
					interval = setInterval(obj.CustomMethod, delay);
					obj.onmouseover = function() {
						clearInterval(interval);
					}
					obj.onmouseout = function() {
						interval = setInterval(obj.CustomMethod, delay);
					}
				}
			}
