    function onKeyEvent(event){
        if (event.keyCode != '13') {
            return;
        }
        checkValues();
    }
    function checkValues(){
		var url = $("#securl").val();
        var keyword = $("input[name='keyword']").val();
		if (keyword === '请输入关键字') {
			keyword = ''; 
		}
		if (keyword !== '') {
			keyword = encodeURI(encodeURI(keyword));
			url = url + "&keyword=" + keyword;
		}
        window.open(url);
    }
