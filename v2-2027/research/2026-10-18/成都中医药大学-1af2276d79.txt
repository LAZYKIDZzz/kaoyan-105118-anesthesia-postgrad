            var effect = 'fade';

            var li = "#" + name + " li";

                $("#" + name + " .li1").first().addClass("on1");
            } //默认第一个加.on1类
            index = $("#" + name + " .li1").index($("#" + name + " .on1"));

            //鼠标离开导航后，回复默认.on1类位置
            $("#" + name)
                .hover(
                    function () {
                        if (navST1 != null) {
                            clearTimeout(navST1);
                        }
                    },
                    function () {
                        navST1 = setTimeout(function () {
                                $("#" + name + " .li1").removeClass("on1").eq(index).addClass("on1");
                            },
                            500);
                    }
                );

            if (type == '1') {
                li = "#" + name + " .li1";
            }
            if (appendItem != '#') { //插入内容
                var appendHtml = $(appendItem).html();
                $(li).first().append(appendHtml);
                $(appendItem).remove();
            }

            if (type == '3') {
                $("#" + name + " .on1").find("ul").first().show();
            }

            $(li)
                .hover(function () {
                        var curItem = $(this);
                        var onNum = (curItem.attr("class").split(" "))[0].replace("li", "");
                        $(li).removeClass("on" + onNum);
                        curItem.addClass("on" + onNum);
                        navST = setTimeout(function () { //延时触发

                                if ($("ul:first", curItem).css("display") != "block") {
                                    $(li + " .ul" + onNum).hide();
                                    if (effect == 'fade') {
                                        $("ul:first", curItem).fadeIn(t);
                                    } else {
                                        $("ul:first", curItem).slideDown(t);
                                    }
                                };
                                navST = null;
                            },
                            t);
                    },
                    function () {
                        if (navST != null) {
                            clearTimeout(navST);
                        }
                        if (type == '1' || type == '2') {
                            if (effect == 'fade') {
                                $(this).find("ul").first().fadeOut(t);
                            } else {
                                $(this).find("ul").first().slideUp(t);
                            }
                        }
                        if (removeOn == 'True') {
                            $(this).removeClass("on1");
                        }
                    },
                    t); //end hover

                            成都中医药大学关于调整2027年硕士研究生招生考试初试科目&科目构成的公告
                             
                         
                         
                             来源：    发布时间：2026-07-15 09:57:15   浏览次数：  次   【字体： 小   大 】 
                         
                         
                         
                             各位考生 :  根据《教育部办公厅关于进一步规范和加强研究生考试招生工作的通知》等有关文件精神，为进一步加强硕士研究生初试考试科目优化调整和规范管理， 结合我校学科发展需要和实际情况，我校 2027年硕士研究生招生考试 以下 专业初试科目 和部份初试科目构成 进行调整，具体如下： 一、中医药信息学（专业代码： 1005Z1）硕士研究生招生考试初试业务课一由自命题科目“694医学信息综合”调整为统考科目“307临床医学综合能力（中医）”。 二、中医药人工智能（专业代码:1005J1）硕士研究生招生考试初试业务课一由自命题科目“694医学信息综合”调整为统考科目“307临床医学综合能力（中医）”。 三、 公共卫生与预防医学（ 专业代码前四位为 1004）硕士研究生招生考试初试业务课一由自命题科目“612预防医学综合 ”调整为 自命题 科目 “ 353卫生综合” （ 科目构成： 卫生统计学 100分，流行病学100分，卫生毒理学100分 ）。 四、公共卫生（专业代码： 105300） 硕士研究生招生考试初试业务课一自命题科目 “ 353卫生综合” 科目构成调整为 卫生统计学 100分，流行病学100分，卫生毒理学100分 。   请广大考生及时关注并积极备考，具体考试科目请以《成都中医药大学 202 7 年硕士研究生招生章程及专业目录》为准。
